create table public.customers (
  id uuid not null default gen_random_uuid (),
  store_id uuid not null,
  name text not null,
  phone text null,
  email text null,
  total_spend numeric(12, 2) not null default 0,
  outstanding_balance numeric(12, 2) not null default 0,
  notes text null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  constraint customers_pkey primary key (id),
  constraint customers_store_id_fkey foreign KEY (store_id) references stores (id) on delete CASCADE,
  constraint customers_outstanding_balance_check check ((outstanding_balance >= (0)::numeric)),
  constraint customers_total_spend_check check ((total_spend >= (0)::numeric))
) TABLESPACE pg_default;

create index IF not exists idx_customers_store_id on public.customers using btree (store_id) TABLESPACE pg_default;

create index IF not exists idx_customers_phone on public.customers using btree (store_id, phone) TABLESPACE pg_default;

create trigger trg_customers_updated_at BEFORE
update on customers for EACH row
execute FUNCTION set_updated_at ();
